```ts
export type AlertType = 'choix_en_attente'|'lot_expire'|'devis_non_depose'|'confirmation_en_attente'|'devis_conteste'|'ecart_suivi'|'retro_a_verser'|'avenant_sans_suite'|'relance_planifiee'|'validation_en_retard'
export type AlertSeverity = 'haute'|'moyenne'|'basse'
export interface B2bAlert { id: string; type: AlertType; severity: AlertSeverity; request_id: string; request_ref: string; lot_id: string | null; lot_label: string | null; engagement_id: string | null; age_days: number; suggested_action: string; call_order: ('demandeur'|'artisan')[]; snoozed_until: string | null }
export function readAlertConfig(config: Record<string, any>): { choiceDays; choiceExpiryDays; quoteDays; quoteStaleDays; confirmDays; confirmCriticalDays; orphanDays }   // réutilise readB2bFlowConfig pour choiceExpiryDays / quoteStaleDays / confirmReminderDays
export interface AlertInput { lots: {id; request_id; category; status; closed_reason: string | null}[]; engagements: {id; lot_id; request_id; status; signed_at; selected_at}[]; submissions: {id; engagement_id; status; created_at}[]; commissionRows: CommissionRow[]; followups: {request_id; next_followup_on: string | null; created_at}[]; requestRefs: Record<string, string> }
export function computeB2bAlerts(input: AlertInput, now: Date, cfg: ReturnType<typeof readAlertConfig>): { alerts: B2bAlert[]; snoozed: B2bAlert[]; counts: { total: number; by_request: Record<string, number>; by_engagement: Record<string, number> } }
export async function loadB2bAlertInput(supabase: any, flow: { quoteBaseCapCents: number | null }): Promise<AlertInput>
```

- choix_en_attente : « Appeler le demandeur pour l'aider à choisir un artisan, puis prévenir l'artisan retenu. »
- lot_expire : « Appeler le demandeur : relancer le dossier ou le clore. »
- devis_non_depose : « Appeler le demandeur pour savoir où en est le devis, puis l'artisan retenu. »

- devis_conteste : « Arbitrer dans Commissions en cours après avoir appelé le demandeur puis l'artisan. »
- ecart_suivi : « Vérifier les montants du dossier (dépôts confirmés / suivi). »
- retro_a_verser : « Enregistrer le versement de la rétro-commission au partenaire. » (ou « Récupérer la rétro-commission versée à tort. » si a_recuperer)

- validation_en_retard : « Appeler le demandeur pour savoir si l'étape « {label} » est atteinte et lui faire valider l'avancement, puis l'artisan. »
