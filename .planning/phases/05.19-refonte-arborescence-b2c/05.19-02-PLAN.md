Example skeleton:
```vue
<script setup lang="ts">
const props = defineProps<{
  selectedCategory: string
}>()

const emit = defineEmits<{
  'update:selected-items': [items: string[]]
}>()

const selectedItems = ref<string[]>([])

function toggleItem(itemId: string) {
  const idx = selectedItems.value.indexOf(itemId)
  if (idx > -1) {
    selectedItems.value.splice(idx, 1)
  } else {
    selectedItems.value.push(itemId)
  }
  emit('update:selected-items', selectedItems.value)
}

// Show alert if only energy-coordinator items selected
const needsCoordinator = computed(() => {
  const energy = ['pac', 'photovoltaique', 'geothermie']
  return selectedItems.value.every(item => energy.includes(item))
})
</script>

<template>
  <div>
    <!-- Gros Œuvre / Second Œuvre tabs for renovation_globale -->
    <div v-if="selectedCategory === 'renovation_globale'" class="space-y-6">
      <div>
        <h3 class="text-lg font-bold mb-4">Gros Œuvre</h3>
        <div class="grid grid-cols-2 gap-3">
          <!-- maconnerie, charpente_couverture, menuiserie_ext -->
        </div>
      </div>
      <div>
        <h3 class="text-lg font-bold mb-4">Second Œuvre</h3>
        <div class="grid grid-cols-2 gap-3">
          <!-- isolation_platrerie, menuiserie_int, plomberie_sanitaire, etc -->
        </div>
      </div>
    </div>

    <!-- Energetique grid + alert -->
    <div v-if="selectedCategory === 'renovation_energetique'" class="space-y-4">
      <div v-if="needsCoordinator" class="p-3 bg-yellow-100 rounded border border-yellow-300 text-sm">
        Certains équipements nécessitent une coordination — nous vous proposerons une entreprise générale
      </div>
      <div class="grid grid-cols-3 gap-3">
        <!-- 9 energetique items -->
      </div>
    </div>

    <!-- Prestations grid -->
    <div v-if="selectedCategory === 'prestations_ciblees'" class="grid grid-cols-2 gap-3">
      <!-- 11 prestations items -->
    </div>
  </div>
</template>
```
