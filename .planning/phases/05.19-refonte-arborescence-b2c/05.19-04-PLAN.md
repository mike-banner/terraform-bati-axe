```typescript
import { describe, it, expect } from 'vitest'
import { COMPATIBILITY_MATRIX } from '~/utils/workTypeMatrix'

describe('Matching Logic — Overlap Rule', () => {
  // Helper function (replicate the logic from leads.index.get.ts)
  function doesLeadMatch(
    proType: 'specialiste' | 'generaliste',
    proCategories: string[],
    selectedItems: string[]
  ): boolean {
    if (proType === 'generaliste') return true

    for (const item of selectedItems) {
      const rule = COMPATIBILITY_MATRIX[item]
      if (!rule) continue

      for (const proCategory of proCategories) {
        if (rule.specialistMatches?.includes(proCategory)) {
          return true
        }
      }
    }
    return false
  }

  it('généraliste receives all leads', () => {
    expect(doesLeadMatch('generaliste', [], [])).toBe(true)
    expect(doesLeadMatch('generaliste', ['maconnerie'], ['electricite'])).toBe(true)
    expect(doesLeadMatch('generaliste', ['any'], ['pac'])).toBe(true)
  })

  it('spécialiste with [maconnerie] receives [maconnerie] lead', () => {
    expect(doesLeadMatch('specialiste', ['maconnerie'], ['maconnerie'])).toBe(true)
  })

  it('spécialiste with [maconnerie] does NOT receive [electricite] lead', () => {
    expect(doesLeadMatch('specialiste', ['maconnerie'], ['electricite'])).toBe(false)
  })

  it('spécialiste with [plomberie, carrelage] receives [salle_de_bain]', () => {
    // salle_de_bain.specialistMatches should include both 'carrelage' and 'plomberie'
    expect(doesLeadMatch('specialiste', ['plomberie', 'carrelage'], ['salle_de_bain'])).toBe(true)
  })

  it('spécialiste with [isolation] does NOT receive [pac] alone (PAC = généraliste only)', () => {
    // PAC.specialistMatches should be empty per D-05
    const pacRule = COMPATIBILITY_MATRIX['pac']
    expect(pacRule.defaultRole).toBe('generaliste')
    expect(doesLeadMatch('specialiste', ['isolation'], ['pac'])).toBe(false)
  })

  it('spécialiste receives nothing if selected_items is empty', () => {
    expect(doesLeadMatch('specialiste', ['maconnerie'], [])).toBe(false)
  })

  it('COMPATIBILITY_MATRIX contains all 26 items', () => {
    const expectedItems = [
      'maconnerie', 'charpente_couverture', 'menuiserie_ext',
      'isolation_platrerie', 'menuiserie_int', 'plomberie_sanitaire', 'revetement_sol', 'peinture_finitions', 'electricite',
      'pac', 'chaudiere_reno', 'borne_irve', 'vmc_chauffage', 'isolation_ite_iti', 'geothermie', 'poele_bois_granules', 'photovoltaique', 'menuiserie_ext_rge',
      'salle_de_bain', 'cuisine', 'toiture', 'facade', 'extension', 'surelevation', 'combles', 'demolition', 'terrasse', 'cloture_portail', 'assainissement'
    ]
    for (const item of expectedItems) {
      expect(COMPATIBILITY_MATRIX[item]).toBeDefined()
      expect(COMPATIBILITY_MATRIX[item].label).toBeTruthy()
      expect(COMPATIBILITY_MATRIX[item].defaultRole).toMatch(/specialiste|generaliste/)
    }
  })
})
```
